### 2. TTS核心服务 (backend/src/services/ttsService.js)
```javascript
import { Communicate } from '@travisvn/edge-tts';

// 音色配置
const VOICE_CONFIG = {
    // 女声：晓晓（适合法官、女巫等角色）
    female: 'zh-CN-XiaoxiaoNeural',
    // 男声：云扬（适合狼人、预言家等角色）
    male: 'zh-CN-YunyangNeural',
    // 年轻男声：云希（适合平民、新手玩家）
    youngMale: 'zh-CN-YunxiNeural',
};

// 语速配置 (默认1.0)
const SPEED_CONFIG = {
    normal: 1.0,
    slow: 0.8,    // 适合重要信息
    fast: 1.2,    // 适合紧张场景
};

class TTSService {
    /**
     * 将文本转换为音频流
     * @param {string} text - 要转换的文本
     * @param {string} voice - 音色标识 (female/male/youngMale)
     * @param {number} speed - 语速 (0.5-2.0)
     * @returns {AsyncGenerator} 音频数据流
     */
    async textToStream(text, voice = 'female', speed = 1.0) {
        try {
            const voiceName = VOICE_CONFIG[voice] || VOICE_CONFIG.female;
            
            // 构建SSML以控制语速和停顿
            const ssml = `
                <speak xmlns="http://www.w3.org/2001/10/synthesis" 
                       xml:lang="zh-CN">
                    <prosody rate="${speed}">
                        ${text}
                    </prosody>
                </speak>
            `;
            
            const communicate = new Communicate(ssml, { 
                voice: voiceName,
                outputFormat: 'audio-24khz-48kbitrate-mono-mp3'
            });
            
            return communicate.stream();
        } catch (error) {
            console.error('TTS转换失败:', error);
            throw new Error('语音合成服务暂时不可用');
        }
    }

    /**
     * 批量转换：将多段文本按角色语音合成
     * @param {Array} segments - [{text, voice}]
     * @returns {Promise<Buffer>} 合并后的音频Buffer
     */
    async batchTextToAudio(segments) {
        const audioBuffers = [];
        
        for (const segment of segments) {
            const stream = await this.textToStream(segment.text, segment.voice);
            const chunks = [];
            
            for await (const chunk of stream) {
                if (chunk.type === 'audio' && chunk.data) {
                    chunks.push(chunk.data);
                }
            }
            
            audioBuffers.push(Buffer.concat(chunks));
        }
        
        // 合并所有音频片段
        return Buffer.concat(audioBuffers);
    }
}

export default new TTSService();
```

### 3. TTS控制器 (backend/src/controllers/ttsController.js)
```javascript
import ttsService from '../services/ttsService.js';

export const ttsController = {
    /**
     * 流式返回音频（推荐：边生成边播放）
     */
    async streamAudio(req, res) {
        try {
            const { text, voice = 'female', speed = 1.0 } = req.body;
            
            if (!text) {
                return res.status(400).json({ error: '缺少文本参数' });
            }

            // 设置响应头
            res.setHeader('Content-Type', 'audio/mpeg');
            res.setHeader('Transfer-Encoding', 'chunked');
            res.setHeader('Cache-Control', 'no-cache');

            const stream = await ttsService.textToStream(text, voice, speed);
            
            // 流式写入响应
            for await (const chunk of stream) {
                if (chunk.type === 'audio' && chunk.data) {
                    res.write(chunk.data);
                }
            }
            
            res.end();
        } catch (error) {
            console.error('流式音频生成失败:', error);
            if (!res.headersSent) {
                res.status(500).json({ error: '语音生成失败' });
            } else {
                res.end();
            }
        }
    },

    /**
     * 批量生成音频（适合完整对局）
     */
    async batchAudio(req, res) {
        try {
            const { segments } = req.body; // [{text, voice}]
            
            if (!segments || !Array.isArray(segments) || segments.length === 0) {
                return res.status(400).json({ error: '缺少语音片段数据' });
            }

            const audioBuffer = await ttsService.batchTextToAudio(segments);
            
            res.setHeader('Content-Type', 'audio/mpeg');
            res.send(audioBuffer);
        } catch (error) {
            console.error('批量音频生成失败:', error);
            res.status(500).json({ error: '批量语音生成失败' });
        }
    }
};
```

## 🎨 前端实现 (Vue 3)

### 1. TTS服务 (frontend/src/services/ttsService.ts)
```typescript
// TTS API调用服务
const API_BASE = 'http://localhost:3000/api';

interface TTSOptions {
    voice?: 'female' | 'male' | 'youngMale';
    speed?: number;
}

export class TTSService {
    private audioContext: AudioContext | null = null;
    private currentSource: AudioBufferSourceNode | null = null;

    /**
     * 流式播放语音（边生成边播放）
     */
    async playStream(text: string, options: TTSOptions = {}) {
        const { voice = 'female', speed = 1.0 } = options;
        
        try {
            const response = await fetch(`${API_BASE}/tts/stream`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ text, voice, speed })
            });

            if (!response.ok) {
                throw new Error('语音生成失败');
            }

            // 获取音频数据
            const audioData = await response.arrayBuffer();
            
            // 使用AudioContext播放
            if (!this.audioContext) {
                this.audioContext = new AudioContext();
            }
            
            // 解码并播放
            const audioBuffer = await this.audioContext.decodeAudioData(audioData);
            const source = this.audioContext.createBufferSource();
            source.buffer = audioBuffer;
            source.connect(this.audioContext.destination);
            source.start();
            
            this.currentSource = source;
            return source;
        } catch (error) {
            console.error('播放失败:', error);
            throw error;
        }
    }

    /**
     * 停止播放
     */
    stopPlay() {
        if (this.currentSource) {
            this.currentSource.stop();
            this.currentSource = null;
        }
    }

    /**
     * 批量播放完整对局
     */
    async playGameScript(script: string, callback?: (segment: string) => void) {
        // 按发言段落分割（根据你的对局格式调整）
        const segments = this.parseScriptToSegments(script);
        
        for (let i = 0; i < segments.length; i++) {
            const seg = segments[i];
            // 根据发言者选择音色（需实现角色-音色映射）
            const voice = this.getVoiceBySpeaker(seg.speaker);
            await this.playStream(seg.text, { voice });
            
            if (callback) {
                callback(seg.text);
            }
            
            // 播放间隔
            await this.sleep(300);
        }
    }

    private parseScriptToSegments(script: string) {
        // 根据你的对局格式解析发言段落
        // 例如：按 "X号玩家发言：" 分割
        const regex = /(\d+号玩家[：:])/g;
        const segments = [];
        let lastIndex = 0;
        let match;
        
        while ((match = regex.exec(script)) !== null) {
            if (lastIndex > 0) {
                const text = script.substring(lastIndex, match.index).trim();
                if (text) {
                    segments.push({
                        speaker: match[1],
                        text: text
                    });
                }
            }
            lastIndex = match.index;
        }
        
        // 处理最后一段
        if (lastIndex < script.length) {
            segments.push({
                speaker: '法官',
                text: script.substring(lastIndex).trim()
            });
        }
        
        return segments;
    }

    private getVoiceBySpeaker(speaker: string): 'female' | 'male' | 'youngMale' {
        // 实现角色到音色的映射逻辑
        // 示例：法官用女声，1-6号用男声，7-12号用年轻男声
        const number = parseInt(speaker);
        if (isNaN(number)) return 'female';
        if (number <= 6) return 'male';
        return 'youngMale';
    }

    private sleep(ms: number) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

export const ttsService = new TTSService();
```

<template>
  <div class="audio-player">
    <!-- 播放控制按钮 -->
    <div class="controls">

        <span v-if="isPlaying">⏸️ 暂停</span>

    </div>

      {{ currentText }}

        <option value="female">晓晓 (女声)</option>
        <option value="male">云扬 (男声)</option>
        <option value="youngMale">云希 (年轻男声)</option>
      </select>
    </div>

    <!-- 语速控制 -->

      <label>语速：{{ speed }}</label>
      <input 
        type="range" 
        min="0.5" 
        max="2.0" 
        step="0.1" 
        v-model="speed"
      />

<script setup lang="ts">
import { ref, computed } from 'vue';
import { ttsService } from '../services/ttsService';

const props = defineProps<{
  text: string;  // 要播放的文本
}>();

const isPlaying = ref(false);
const isLoading = ref(false);
const progress = ref(0);
const currentText = ref('');
const selectedVoice = ref<'female' | 'male' | 'youngMale'>('female');
const speed = ref(1.0);

const togglePlay = async () => {
  if (isPlaying.value) {
    ttsService.stopPlay();
    isPlaying.value = false;
    return;
  }

  try {
    isLoading.value = true;
    currentText.value = props.text;
    
    await ttsService.playStream(props.text, {
      voice: selectedVoice.value,
      speed: speed.value
    });
    
    isPlaying.value = true;
    // 模拟进度（实际可从AudioContext获取）
    progress.value = 100;
  } catch (error) {
    console.error('播放失败:', error);
  } finally {
    isLoading.value = false;
    isPlaying.value = false;
  }
};

const stopPlay = () => {
  ttsService.stopPlay();
  isPlaying.value = false;
  progress.value = 0;
};
</script>

## ⚠️ 注意事项

1. **CORS配置**：确保后端正确配置了CORS，允许前端域名访问
2. **音频格式**：Edge TTS默认输出MP3格式，确保浏览器支持
3. **长文本处理**：单次请求建议控制在5000字以内，超出需分段
4. **并发限制**：Edge TTS有并发限制，建议添加请求队列
5. **网络问题**：Edge TTS需要访问微软服务，确保服务器有外网访问权限
